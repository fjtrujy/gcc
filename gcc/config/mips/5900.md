;; -------------------------------------------------------------------------
;; FPU (COP1) Accumulator Register Operations for R5900
;; The R5900 FPU has a 32-bit accumulator register (ACC) used for
;; 3-operand FMA operations (not the 4-operand MADD4 form).
;; -------------------------------------------------------------------------

;; FPU ACC register constant
(define_constants
  [(FPU_ACC_REGNUM		189)])

;; FPU ACC UNSPEC constants
(define_c_enum "unspec" [
  ;; ACC-writing instructions
  UNSPEC_FPU_ADDA       ;; ACC = fs + ft
  UNSPEC_FPU_SUBA       ;; ACC = fs - ft
  UNSPEC_FPU_MULA       ;; ACC = fs * ft
  UNSPEC_FPU_MADDA      ;; ACC = ACC + fs * ft
  UNSPEC_FPU_MSUBA      ;; ACC = ACC - fs * ft
  ;; ACC-reading instructions
  UNSPEC_FPU_MADD       ;; fd = ACC + fs * ft
  UNSPEC_FPU_MSUB       ;; fd = ACC - fs * ft
])

;; adda.s: ACC = fs + ft
(define_insn "fpu_adda"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_ADDA))]
  "TARGET_MIPS5900"
  "adda.s\t%0,%1"
  [(set_attr "type" "fadd")
   (set_attr "mode" "SF")])

;; suba.s: ACC = fs - ft
(define_insn "fpu_suba"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_SUBA))]
  "TARGET_MIPS5900"
  "suba.s\t%0,%1"
  [(set_attr "type" "fadd")
   (set_attr "mode" "SF")])

;; mula.s: ACC = fs * ft
(define_insn "fpu_mula"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_MULA))]
  "TARGET_MIPS5900"
  "mula.s\t%0,%1"
  [(set_attr "type" "fmul")
   (set_attr "mode" "SF")])

;; madda.s: ACC = ACC + fs * ft
(define_insn "fpu_madda"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 3 "fpu_acc_operand" "2")
                    (match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_MADDA))]
  "TARGET_MIPS5900"
  "madda.s\t%0,%1"
  [(set_attr "type" "fmadd")
   (set_attr "mode" "SF")])

;; msuba.s: ACC = ACC - fs * ft
(define_insn "fpu_msuba"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 3 "fpu_acc_operand" "2")
                    (match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_MSUBA))]
  "TARGET_MIPS5900"
  "msuba.s\t%0,%1"
  [(set_attr "type" "fmadd")
   (set_attr "mode" "SF")])

;; madd.s: fd = ACC + fs * ft (reads ACC, writes to FP register)
(define_insn "fpu_madd"
  [(set (match_operand:SF 0 "register_operand" "=f")
        (unspec:SF [(match_operand:SF 3 "fpu_acc_operand" "YF")
                    (match_operand:SF 1 "register_operand" "f")
                    (match_operand:SF 2 "register_operand" "f")]
                   UNSPEC_FPU_MADD))]
  "TARGET_MIPS5900"
  "madd.s\t%0,%1,%2"
  [(set_attr "type" "fmadd")
   (set_attr "mode" "SF")])

;; msub.s: fd = ACC - fs * ft (reads ACC, writes to FP register)
(define_insn "fpu_msub"
  [(set (match_operand:SF 0 "register_operand" "=f")
        (unspec:SF [(match_operand:SF 3 "fpu_acc_operand" "YF")
                    (match_operand:SF 1 "register_operand" "f")
                    (match_operand:SF 2 "register_operand" "f")]
                   UNSPEC_FPU_MSUB))]
  "TARGET_MIPS5900"
  "msub.s\t%0,%1,%2"
  [(set_attr "type" "fmadd")
   (set_attr "mode" "SF")])

;; -------------------------------------------------------------------------
;; FPU ACC Move Patterns (for reload/spill)
;; These synthetic sequences allow moving values to/from FPU ACC.
;; -------------------------------------------------------------------------

;; Move SF value to FPU ACC: ACC = src + 0.0
;; Uses mtc1 to load zero, then adda.s to add src + 0 = src
(define_insn "movsf_to_fpu_acc"
  [(set (match_operand:SF 0 "fpu_acc_operand" "=YF")
        (match_operand:SF 1 "register_operand" "f"))
   (clobber (match_scratch:SF 2 "=&f"))]
  "TARGET_MIPS5900"
  "mtc1\t$0,%2\;adda.s\t%1,%2"
  [(set_attr "type" "multi")
   (set_attr "mode" "SF")
   (set_attr "length" "8")])

;; Move FPU ACC to SF value: dst = ACC + 0.0*0.0 = ACC
;; Uses mtc1 to load zero, then madd.s: dst = ACC + 0*0 = ACC
(define_insn "movsf_from_fpu_acc"
  [(set (match_operand:SF 0 "register_operand" "=f")
        (match_operand:SF 1 "fpu_acc_operand" "YF"))
   (clobber (match_scratch:SF 2 "=&f"))]
  "TARGET_MIPS5900"
  "mtc1\t$0,%2\;madd.s\t%0,%2,%2"
  [(set_attr "type" "multi")
   (set_attr "mode" "SF")
   (set_attr "length" "8")])

;; -------------------------------------------------------------------------
;; R5900 Pipeline 1 Multiply/Divide Instructions
;; The R5900 has a second multiply/divide unit (MAC1) in Pipeline 1
;; using dedicated HI1/LO1 registers.
;; -------------------------------------------------------------------------

;; Pipeline 1 register constant (HI1_REGNUM and LO1_REGNUM defined in mips.h)
(define_constants
  [(MD1_REG_FIRST	190)])

;; UNSPEC constants for Pipeline 1 operations
(define_c_enum "unspec" [
  UNSPEC_MULT1		;; MULT1: HI1:LO1 = rs * rt (signed)
  UNSPEC_MULTU1		;; MULTU1: HI1:LO1 = rs * rt (unsigned)
  UNSPEC_DIV1		;; DIV1: LO1 = rs / rt, HI1 = rs % rt (signed)
  UNSPEC_DIVU1		;; DIVU1: LO1 = rs / rt, HI1 = rs % rt (unsigned)
  UNSPEC_MADD1		;; MADD1: HI1:LO1 += rs * rt (signed)
  UNSPEC_MADDU1		;; MADDU1: HI1:LO1 += rs * rt (unsigned)
  UNSPEC_MFHI1		;; MFHI1: rd = HI1
  UNSPEC_MFLO1		;; MFLO1: rd = LO1
  UNSPEC_MTHI1		;; MTHI1: HI1 = rs
  UNSPEC_MTLO1		;; MTLO1: LO1 = rs
])

;; -------------------------------------------------------------------------
;; Pipeline 1 Scheduling
;; MAC1 can execute in parallel with MAC0 for improved throughput.
;; -------------------------------------------------------------------------

;; CPU unit for Pipeline 1 multiply/divide
(define_cpu_unit "r5900_mac1" "r5900_alu")

;; MFHI1/MFLO1/MTHI1/MTLO1 - Pipeline 1 HI1/LO1 moves
(define_insn_reservation "r5900_hilo1" 1
  (and (eq_attr "cpu" "r5900")
       (eq_attr "type" "mfhi,mflo,mthi,mtlo")
       (match_test "MD1_REG_P (REGNO (SET_DEST (PATTERN (insn))))
		    || (GET_CODE (SET_SRC (PATTERN (insn))) == REG
			&& MD1_REG_P (REGNO (SET_SRC (PATTERN (insn)))))"))
  "r5900_mac1")

;; MULT1/MULTU1/MADD1/MADDU1 - Pipeline 1 multiply (4 cycles, 2 repeat)
(define_insn_reservation "r5900_imul1" 4
  (and (eq_attr "cpu" "r5900")
       (eq_attr "type" "imul,imul3,imadd")
       (match_test "MD1_REG_P (REGNO (SET_DEST (PATTERN (insn))))"))
  "r5900_mac1*2,nothing*2")

;; DIV1/DIVU1 - Pipeline 1 divide (37 cycles)
(define_insn_reservation "r5900_idiv1" 37
  (and (eq_attr "cpu" "r5900")
       (eq_attr "type" "idiv,idiv3")
       (match_test "MD1_REG_P (REGNO (SET_DEST (PATTERN (insn))))"))
  "r5900_mac1*37")

;; -------------------------------------------------------------------------
;; Pipeline 1 Instruction Patterns
;; -------------------------------------------------------------------------

;; MULT1 - Multiply Word Pipeline 1 (signed)
;; (HI1, LO1) = rs * rt
(define_insn "mult1_si"
  [(set (match_operand:DI 0 "hilo1_operand" "=Ym")
        (mult:DI (sign_extend:DI (match_operand:SI 1 "register_operand" "d"))
                 (sign_extend:DI (match_operand:SI 2 "register_operand" "d"))))]
  "TARGET_MIPS5900"
  "mult1\t%1,%2"
  [(set_attr "type" "imul")
   (set_attr "mode" "SI")])

;; MULTU1 - Multiply Unsigned Word Pipeline 1
;; (HI1, LO1) = rs * rt (unsigned)
(define_insn "multu1_si"
  [(set (match_operand:DI 0 "hilo1_operand" "=Ym")
        (mult:DI (zero_extend:DI (match_operand:SI 1 "register_operand" "d"))
                 (zero_extend:DI (match_operand:SI 2 "register_operand" "d"))))]
  "TARGET_MIPS5900"
  "multu1\t%1,%2"
  [(set_attr "type" "imul")
   (set_attr "mode" "SI")])

;; DIV1 - Divide Word Pipeline 1 (signed)
;; LO1 = rs / rt, HI1 = rs % rt
(define_insn "div1_si"
  [(set (match_operand:DI 0 "hilo1_operand" "=Ym")
        (unspec:DI [(match_operand:SI 1 "register_operand" "d")
                    (match_operand:SI 2 "register_operand" "d")]
                   UNSPEC_DIV1))]
  "TARGET_MIPS5900"
  "div1\t$0,%1,%2"
  [(set_attr "type" "idiv")
   (set_attr "mode" "SI")])

;; DIVU1 - Divide Unsigned Word Pipeline 1
(define_insn "divu1_si"
  [(set (match_operand:DI 0 "hilo1_operand" "=Ym")
        (unspec:DI [(match_operand:SI 1 "register_operand" "d")
                    (match_operand:SI 2 "register_operand" "d")]
                   UNSPEC_DIVU1))]
  "TARGET_MIPS5900"
  "divu1\t$0,%1,%2"
  [(set_attr "type" "idiv")
   (set_attr "mode" "SI")])

;; MADD1 - Multiply-Add Word Pipeline 1 (signed)
;; (HI1, LO1) = (HI1, LO1) + rs * rt
(define_insn "madd1_si"
  [(set (match_operand:DI 0 "hilo1_operand" "=Ym")
        (plus:DI
          (mult:DI (sign_extend:DI (match_operand:SI 1 "register_operand" "d"))
                   (sign_extend:DI (match_operand:SI 2 "register_operand" "d")))
          (match_operand:DI 3 "hilo1_operand" "0")))]
  "TARGET_MIPS5900"
  "madd1\t%1,%2"
  [(set_attr "type" "imadd")
   (set_attr "mode" "SI")])

;; MADDU1 - Multiply-Add Unsigned Word Pipeline 1
(define_insn "maddu1_si"
  [(set (match_operand:DI 0 "hilo1_operand" "=Ym")

  "TARGET_MIPS5900"
  "maddu1\t%1,%2"
