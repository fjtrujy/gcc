;; RSQRT
(define_insn "rsqrtsf"
  [(set (match_operand:SF 0 "register_operand" "=f")
	(div:SF (match_operand:SF 1 "register_operand" "f")
		  (sqrt:SF (match_operand:SF 2 "register_operand" "f"))))]
  "TARGET_MIPS5900"
  "rsqrt.s\t%0,%1,%2"
  [(set_attr "type" "frsqrt")
   (set_attr "mode" "SF")])

;; -------------------------------------------------------------------------
;; FPU (COP1) Accumulator Register Operations for R5900
;; The R5900 FPU has a 32-bit accumulator register (ACC) used for
;; 3-operand FMA operations (not the 4-operand MADD4 form).
;; -------------------------------------------------------------------------

;; FPU ACC register constant
(define_constants
  [(FPU_ACC_REGNUM		188)])

;; FPU ACC UNSPEC constants
(define_c_enum "unspec" [
  ;; ACC-writing instructions
  UNSPEC_FPU_ADDA       ;; ACC = fs + ft
  UNSPEC_FPU_SUBA       ;; ACC = fs - ft
  UNSPEC_FPU_MULA       ;; ACC = fs * ft
  UNSPEC_FPU_MADDA      ;; ACC = ACC + fs * ft
  UNSPEC_FPU_MSUBA      ;; ACC = ACC - fs * ft
  ;; ACC-reading instructions
  UNSPEC_FPU_MADD       ;; fd = ACC + fs * ft
  UNSPEC_FPU_MSUB       ;; fd = ACC - fs * ft
])

;; adda.s: ACC = fs + ft
(define_insn "fpu_adda"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_ADDA))]
  "TARGET_MIPS5900"
  "adda.s\t%0,%1"
  [(set_attr "type" "fadd")
   (set_attr "mode" "SF")])

;; suba.s: ACC = fs - ft
(define_insn "fpu_suba"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_SUBA))]
  "TARGET_MIPS5900"
  "suba.s\t%0,%1"
  [(set_attr "type" "fadd")
   (set_attr "mode" "SF")])

;; mula.s: ACC = fs * ft
(define_insn "fpu_mula"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_MULA))]
  "TARGET_MIPS5900"
  "mula.s\t%0,%1"
  [(set_attr "type" "fmul")
   (set_attr "mode" "SF")])

;; madda.s: ACC = ACC + fs * ft
(define_insn "fpu_madda"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 3 "fpu_acc_operand" "2")
                    (match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_MADDA))]
  "TARGET_MIPS5900"
  "madda.s\t%0,%1"
  [(set_attr "type" "fmadd")
   (set_attr "mode" "SF")])

;; msuba.s: ACC = ACC - fs * ft
(define_insn "fpu_msuba"
  [(set (match_operand:SF 2 "fpu_acc_operand" "=YF")
        (unspec:SF [(match_operand:SF 3 "fpu_acc_operand" "2")
                    (match_operand:SF 0 "register_operand" "f")
                    (match_operand:SF 1 "register_operand" "f")]
                   UNSPEC_FPU_MSUBA))]
  "TARGET_MIPS5900"
  "msuba.s\t%0,%1"
  [(set_attr "type" "fmadd")
   (set_attr "mode" "SF")])

;; madd.s: fd = ACC + fs * ft (reads ACC, writes to FP register)
(define_insn "fpu_madd"
  [(set (match_operand:SF 0 "register_operand" "=f")
        (unspec:SF [(match_operand:SF 3 "fpu_acc_operand" "YF")
                    (match_operand:SF 1 "register_operand" "f")
                    (match_operand:SF 2 "register_operand" "f")]
                   UNSPEC_FPU_MADD))]
  "TARGET_MIPS5900"
  "madd.s\t%0,%1,%2"
  [(set_attr "type" "fmadd")
   (set_attr "mode" "SF")])

;; msub.s: fd = ACC - fs * ft (reads ACC, writes to FP register)
(define_insn "fpu_msub"
  [(set (match_operand:SF 0 "register_operand" "=f")
        (unspec:SF [(match_operand:SF 3 "fpu_acc_operand" "YF")
                    (match_operand:SF 1 "register_operand" "f")
                    (match_operand:SF 2 "register_operand" "f")]
                   UNSPEC_FPU_MSUB))]
  "TARGET_MIPS5900"
  "msub.s\t%0,%1,%2"
  [(set_attr "type" "fmadd")
   (set_attr "mode" "SF")])

;; -------------------------------------------------------------------------
;; FPU ACC Move Patterns (for reload/spill)
;; These synthetic sequences allow moving values to/from FPU ACC.
;; -------------------------------------------------------------------------

;; Move SF value to FPU ACC: ACC = src + 0.0
;; Uses mtc1 to load zero, then adda.s to add src + 0 = src
(define_insn "movsf_to_fpu_acc"
  [(set (match_operand:SF 0 "fpu_acc_operand" "=YF")
        (match_operand:SF 1 "register_operand" "f"))
   (clobber (match_scratch:SF 2 "=&f"))]
  "TARGET_MIPS5900"
  "mtc1\t$0,%2\;adda.s\t%1,%2"
  [(set_attr "type" "multi")
   (set_attr "mode" "SF")
   (set_attr "length" "8")])

;; Move FPU ACC to SF value: dst = ACC + 0.0*0.0 = ACC
;; Uses mtc1 to load zero, then madd.s: dst = ACC + 0*0 = ACC
(define_insn "movsf_from_fpu_acc"
  [(set (match_operand:SF 0 "register_operand" "=f")
        (match_operand:SF 1 "fpu_acc_operand" "YF"))
   (clobber (match_scratch:SF 2 "=&f"))]
  "TARGET_MIPS5900"
  "mtc1\t$0,%2\;madd.s\t%0,%2,%2"
  [(set_attr "type" "multi")
   (set_attr "mode" "SF")
   (set_attr "length" "8")])
