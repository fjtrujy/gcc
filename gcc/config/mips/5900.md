;; R5900 instruction patterns and pipeline tuning.

(define_automaton "r5900_alu,r5900_fpu,r5900_br,r5900_ls,r5900_mac")

(define_cpu_unit "r5900_alu0,r5900_alu1" "r5900_alu")

(define_cpu_unit "r5900_br" "r5900_br")
(define_cpu_unit "r5900_ls" "r5900_ls")
;; R5900 has two MAC units: MAC0 (Pipeline 0) and MAC1 (Pipeline 1)
(define_cpu_unit "r5900_mac0,r5900_mac1" "r5900_mac")

(define_insn_reservation "r5900_alu" 1
  (and (eq_attr "cpu" "r5900")
       (eq_attr "type" "unknown,prefetch,prefetchx,condmove,const,arith,

 "r5900_alu0|r5900_alu1")

;; RSQRT
(define_insn "rsqrtsf"
  [(set (match_operand:SF 0 "register_operand" "=f")
	(div:SF (match_operand:SF 1 "register_operand" "f")
		  (sqrt:SF (match_operand:SF 2 "register_operand" "f"))))]
  "TARGET_MIPS5900"
  "rsqrt.s\t%0,%1,%2"
  [(set_attr "type" "frsqrt")
   (set_attr "mode" "SF")])

;; Define constants for R5900-specific unspecs
(define_c_enum "unspec" [
  UNSPEC_R5900_MULT1
  UNSPEC_R5900_MULTU1
  UNSPEC_R5900_DIV1
  UNSPEC_R5900_DIVU1
  UNSPEC_R5900_MADD1
  UNSPEC_R5900_MADDU1
  UNSPEC_R5900_MFHI1
  UNSPEC_R5900_MFLO1
  UNSPEC_R5900_MTHI1
  UNSPEC_R5900_MTLO1
])

;; MULT1 - Signed multiply using Pipeline 1
;; 2-operand form: HI1:LO1 = rs * rt
;; 3-operand form: rd = LO1 = rs * rt (low 32 bits), HI1 = high 32 bits
(define_insn "r5900_mult1"
  [(set (match_operand:DI 0 "register_operand" "=Ym")
	(unspec:DI [(match_operand:SI 1 "register_operand" "d")
		    (match_operand:SI 2 "register_operand" "d")]
		   UNSPEC_R5900_MULT1))]
  "TARGET_MIPS5900"
  "mult1\t%1,%2"
  [(set_attr "type" "imul1")
   (set_attr "mode" "SI")])

;; MULT1 3-operand form - result goes to GPR and HI1:LO1
(define_insn "r5900_mult1_3op"
  [(set (match_operand:SI 0 "register_operand" "=d")
	(unspec:SI [(match_operand:SI 1 "register_operand" "d")
		    (match_operand:SI 2 "register_operand" "d")]
		   UNSPEC_R5900_MULT1))
   (clobber (match_scratch:DI 3 "=Ym"))]
  "TARGET_MIPS5900"
  "mult1\t%0,%1,%2"
  [(set_attr "type" "imul1")
   (set_attr "mode" "SI")])

;; MULTU1 - Unsigned multiply using Pipeline 1
;; 2-operand form: HI1:LO1 = rs * rt
;; 3-operand form: rd = LO1 = rs * rt (low 32 bits), HI1 = high 32 bits

	(unspec:DI [(match_operand:SI 1 "register_operand" "d")

;; MADD1 - Signed multiply-add using Pipeline 1 (HI1:LO1 += rs * rt)
;; 2-operand form: HI1:LO1 += rs * rt

  [(set (match_operand:DI 0 "register_operand" "=Ym")
	(unspec:DI [(match_operand:DI 1 "register_operand" "0")

		    (match_operand:SI 3 "register_operand" "d")]

  "TARGET_MIPS5900"
  "madd1\t%2,%3"
  [(set_attr "type" "imul1")
   (set_attr "mode" "SI")])

  "TARGET_MIPS5900"
